│ 📦 SNIPPET 1: Parse Slack Mentions                                          │
│ Node Type: Code (JavaScript)                                                │
│ Node Name: "Parse Slack Mentions"                                           │
└──────────────────────────────────────────────────────────────────────────────┘

// Extract all Slack user mentions from message
const slackMessage = $input.item.json.text || '';

// Regex to match Slack mention format: <@U8NC9D5AM>
const mentionRegex = /<@([A-Z0-9]+)>/g;
const mentions = [];
let match;

while ((match = mentionRegex.exec(slackMessage)) !== null) {
  mentions.push({
    fullMatch: match[0],      // "<@U8NC9D5AM>"
    slackUserId: match[1],    // "U8NC9D5AM"
    position: match.index
  });
}

// Also extract display names (like @spencer without the <>)
const displayNameRegex = /(?:^|\s)@(\w+)/g;
const displayNameMentions = [];
let nameMatch;

while ((nameMatch = displayNameRegex.exec(slackMessage)) !== null) {
  displayNameMentions.push({
    fullMatch: nameMatch[0],
    displayName: nameMatch[1],
    position: nameMatch.index
  });
}

return [{
  json: {
    originalMessage: slackMessage,
    slackMentions: mentions,
    displayNameMentions: displayNameMentions,
    hasMentions: mentions.length > 0 || displayNameMentions.length > 0,
    mentionCount: mentions.length,
    displayNameCount: displayNameMentions.length
  }
}];

/* OUTPUT EXAMPLE:
{
  "originalMessage": "assign to <@U8NC9D5AM> and notify @spencer",
  "slackMentions": [
    {
      "fullMatch": "<@U8NC9D5AM>",
      "slackUserId": "U8NC9D5AM",
      "position": 10
    }
  ],
  "displayNameMentions": [
    {
      "fullMatch": "@spencer",
      "displayName": "spencer",
      "position": 35
    }
  ],
  "hasMentions": true,
  "mentionCount": 1,
  "displayNameCount": 1
}
*/

// Get all Slack user IDs from previous node
const mentions = $input.item.json.slackMentions || [];
const slackUserIds = mentions.map(m => m.slackUserId);

if (slackUserIds.length === 0) {
  return [{
    json: {
      resolvedUsers: [],
      unresolvedIds: [],
      allResolved: true
    }
  }];
}

// Query Supabase for all users at once
const { data: users, error } = await $supabase
  .from('gorgias_users')
  .select('slack_user_id, email, full_name, slack_display_name, role')
  .in('slack_user_id', slackUserIds);

if (error) {
  throw new Error(`Supabase query failed: ${error.message}`);
}

// Create lookup map
const resolvedUsers = {};
const resolvedIds = new Set();

users.forEach(user => {
  resolvedUsers[user.slack_user_id] = {
    slackUserId: user.slack_user_id,
    email: user.email,
    fullName: user.full_name,
    displayName: user.slack_display_name,
    role: user.role
  };
  resolvedIds.add(user.slack_user_id);
});

// Find any IDs that didn't resolve
const unresolvedIds = slackUserIds.filter(id => !resolvedIds.has(id));

return [{
  json: {
    originalMessage: $input.item.json.originalMessage,
    resolvedUsers: resolvedUsers,
    unresolvedIds: unresolvedIds,
    allResolved: unresolvedIds.length === 0,
    resolutionCount: Object.keys(resolvedUsers).length,
    totalMentions: slackUserIds.length
  }
}];

┌──────────────────────────────────────────────────────────────────────────────┐
│ 📦 SNIPPET 4: Resolve Display Names to Email                               │
│ Node Type: Code (JavaScript)                                                │
│ Node Name: "Resolve Display Names"                                          │
└──────────────────────────────────────────────────────────────────────────────┘

// Get display name mentions (like @spencer)
const displayNameMentions = $input.item.json.displayNameMentions || [];

if (displayNameMentions.length === 0) {
  return [{
    json: {
      resolvedByName: {},
      unresolvedNames: []
    }
  }];
}

const displayNames = displayNameMentions.map(m => m.displayName);

// Query Supabase with fuzzy matching
const { data: users, error } = await $supabase
  .from('gorgias_users')
  .select('slack_user_id, email, full_name, slack_display_name')
  .or(
    displayNames.map(name => 
      `slack_display_name.ilike.%${name}%,full_name.ilike.%${name}%`
    ).join(',')
  );

if (error) {
  throw new Error(`Supabase query failed: ${error.message}`);
}

// Match display names to users (take first match for each)
const resolvedByName = {};
const unresolvedNames = [];

displayNames.forEach(name => {
  const match = users.find(u => 
    u.slack_display_name?.toLowerCase().includes(name.toLowerCase()) ||
    u.full_name?.toLowerCase().includes(name.toLowerCase())
  );
  
  if (match) {
    resolvedByName[name] = {
      email: match.email,
      fullName: match.full_name,
      displayName: match.slack_display_name,
      slackUserId: match.slack_user_id
    };
  } else {
    unresolvedNames.push(name);
  }
});

return [{
  json: {
    resolvedByName: resolvedByName,
    unresolvedNames: unresolvedNames,
    allNamesResolved: unresolvedNames.length === 0
  }
}];

// Merge into single array of emails

    userDetails.push({
      email: user.email,
      fullName: user.fullName,
      displayName: user.displayName,
      slackUserId: user.slackUserId,
      resolvedVia: 'slack_id'
    });
  }
});

// Add display name resolutions
Object.entries(resolvedByName).forEach(([name, user]) => {
  if (user.email && !allEmails.has(user.email)) {
    allEmails.add(user.email);
    userDetails.push({
      email: user.email,
      fullName: user.fullName,
      displayName: user.displayName,
      slackUserId: user.slackUserId,
      resolvedVia: 'display_name',
      matchedName: name
    });
  }
});

return [{
  json: {
    allEmails: Array.from(allEmails),
    userDetails: userDetails,
    totalResolved: allEmails.size,
    originalMessage: $input.item.json.originalMessage
  }
}];

// Comprehensive error handling for mention resolution failures

const slackUserId = $input.item.json.slackUserId;
const displayName = $input.item.json.displayName;
const originalMessage = $input.item.json.originalMessage;

let email = null;
let errorType = null;
let errorMessage = null;

try {
  // Try primary method: Slack ID lookup
  if (slackUserId) {
    const { data: users, error } = await $supabase
      .from('gorgias_users')
      .select('email, full_name')
      .eq('slack_user_id', slackUserId)
      .limit(1);
    
    if (error) throw new Error(`Database error: ${error.message}`);
    
    if (users && users.length > 0) {
      email = users[0].email;
    }
  }
  
  // Fallback method: Display name lookup
  if (!email && displayName) {
    const { data: users, error } = await $supabase
      .from('gorgias_users')
      .select('email, full_name')
      .ilike('slack_display_name', `%${displayName}%`)
      .limit(1);
    
    if (error) throw new Error(`Database error: ${error.message}`);
    
    if (users && users.length > 0) {
      email = users[0].email;
    }
  }
  
  // Still no match? Log for investigation
  if (!email) {
    errorType = 'user_not_found';
    errorMessage = `No match for Slack ID: ${slackUserId} or name: ${displayName}`;
    
    // Log to Supabase for investigation
    await $supabase.from('resolution_errors').insert([{
      slack_user_id: slackUserId,
      display_name: displayName,
      original_message: originalMessage,
      error_type: errorType,
      error_message: errorMessage
    }]);
  }
  
} catch (error) {
  errorType = 'exception';
  errorMessage = error.message;
  
  console.error('Resolution error:', error);
}

return [{
  json: {
    slackUserId: slackUserId,
    displayName: displayName,
    email: email,
    resolved: !!email,
    errorType: errorType,
    errorMessage: errorMessage,
    originalMessage: originalMessage
  }
}];
